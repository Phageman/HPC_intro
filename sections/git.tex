\section{Git}
\begin{frame}{Git keeps track of every version of your code}
\begin{columns}[c]
    \column{0.65\textwidth}
    \begin{itemize}
        \item Git remembers every saved version of your files
        \item Each saved version is a snapshot, called a \emph{commit}
        \item Go back to any earlier version
        \item See who changed what, when and why
        \item Work together on the same code without overwriting each other's work
    \end{itemize}
    \column{0.35\textwidth}
    \begin{center}
        \includegraphics[height=1.2cm]{logo/github.png} \hspace{0.3cm}
        \includegraphics[height=1.4cm]{logo/gitlab.png}
    \end{center}
    \centering\footnotesize
    GitHub and GitLab store Git projects online
\end{columns}
\end{frame}

\begin{frame}{Git words you need first}
\small
\renewcommand{\arraystretch}{1.25}
\begin{cmdtablex}{lL}
    \textbf{Term}     & \textbf{What it means}                                                       \\
    \midrule
    Repository (repo) & A project directory whose history Git keeps track of                        \\
    Local repo        & The copy on your own computer or the HPC, where you work                     \\
    Remote repo       & The copy online, on GitHub or GitLab; your clone calls it \cmd{origin}       \\
    Clone             & Download a repository to your computer: this makes your local repo           \\
    Fork              & Your own online copy of someone else's repository, which you are free to change \\
\end{cmdtablex}
\bigskip
\hint{Not your repository? Fork it on GitHub first, then clone your fork.}
\end{frame}

\begin{frame}{Where your work lives}
\begin{columns}[c,onlytextwidth]

    \scriptsize

    \begin{cmdtablex}{@{}>{\raggedright\arraybackslash}p{1.3cm}L}
        \textbf{Box}         & \textbf{What it is}                       \\
        \midrule
        Working tree         & Your files, as you edit them              \\
        Staging area         & Changes picked for the next commit        \\
        Local branch         & Your commits on your computer             \\
        Remote-tracking ref  & Git's note of the remote at the last download \\
        Remote branch        & The commits on GitHub                     \\
    \end{cmdtablex}
    \medskip
    A \emph{branch} is one line of history. The main one is \cmd{main}
    (\cmd{master} in older repos).
\end{columns}
\end{frame}

\begin{frame}{Moving your work between them}
\begin{columns}[c,onlytextwidth]
    \column{0.57\textwidth}
    \includegraphics[width=\linewidth]{git_workflow.png}
    \column{0.4\textwidth}
    \scriptsize
    \renewcommand{\arraystretch}{1.2}
    \begin{cmdtablex}{@{}lL}
        \textbf{git \ldots} & \textbf{What it does}            \\
        \midrule
        \cmd{add}        & Stage changes                       \\
        \cmd{commit}     & Save staged changes as a snapshot   \\
        \cmd{push}       & Upload your commits                 \\
        \cmd{pull}       & Download commits into your files    \\
        \midrule
        \cmd{fetch}      & Download, but leave your files      \\
        \cmd{checkout}   & Switch to another branch            \\
        \cmd{merge}      & Combine a branch into yours         \\
    \end{cmdtablex}
    \medskip
    Today you need only the top four.
\end{columns}
\end{frame}

\begin{frame}[fragile]{The basic Git workflow}
Every working day: pull, edit, add, commit, push
\begin{lstlisting}
git clone git@github.com:<user>/<repo>.git
cd <repo>

git pull                # get the latest changes
# ... edit your files ...
git status              # what has changed?
git add <file>          # stage changes for the commit
git commit -m "Describe what you changed"
git push                # upload your commits
\end{lstlisting}
\hint{Cloning with \cmd{git@github.com:} needs your SSH key (exercise 2.4).
    Without it, use the \cmd{https://} URL}
\end{frame}

\begin{frame}[fragile]{Useful Git commands for the exercises}
\small
\begin{cmdtablex}{lL}
    \textbf{Command}                          & \textbf{What it does}                       \\
    \midrule
    \cmd{git config --global user.name}       & Tell Git who you are (once per computer)    \\
    \cmd{git diff}                            & Show changes you have not staged yet        \\
    \cmd{git log --oneline}                   & List earlier commits                        \\
    \cmd{git restore <file>}                  & Undo unstaged changes to a file             \\
    \cmd{git remote -v}                       & Show where your clone pushes and pulls      \\
    \cmd{git remote set-url origin <url>}     & Point your clone at another remote          \\
\end{cmdtablex}
\end{frame}
