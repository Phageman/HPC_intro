\section{Git}
\begin{frame}{Git is your lab journal for code}
\emph{Every entry dated, signed, and never torn out}

\medskip
\begin{columns}[c]
    \column{0.65\textwidth}
    \begin{itemize}
        \item No more \cmd{protocol_v3_FINAL_real.docx}:
              one file, with its whole history
        \item Each entry is a \emph{commit}: what changed, who, when and why
        \item Go back to any earlier version
        \item Work together on the same code without overwriting each other's work
    \end{itemize}
    \column{0.35\textwidth}
    \begin{center}
        \includegraphics[height=1.2cm]{logo/github.png} \hspace{0.3cm}
        \includegraphics[height=1.4cm]{logo/gitlab.png}
    \end{center}
    \centering\footnotesize
    GitHub and GitLab store Git projects online
\end{columns}
\end{frame}

\begin{frame}{Git words you need first}
\small
\renewcommand{\arraystretch}{1.25}
\begin{cmdtablex}{lL}
    \textbf{Term}     & \textbf{What it means}                                                       \\
    \midrule
    Repository (repo) & A project directory whose history Git keeps track of                        \\
    Local repo        & The copy on your own computer or the HPC, where you work                     \\
    Remote repo       & The copy online, on GitHub or GitLab; your clone calls it \cmd{origin}       \\
    Clone             & Download a repository to your computer: this makes your local repo           \\
    Fork              & Your own online copy of someone else's repository, which you are free to change \\
\end{cmdtablex}
\end{frame}

\begin{frame}{Where your work lives}
\diagram{git-areas}

\bigskip
\begin{itemize}
    \item Git saves your work in steps: pick the changes, save them, share them
    \item Each commit is a journal entry. Your local repo holds all of them
    \item A \emph{branch} is one line of history. The main one is \cmd{main}
          (\cmd{master} in older repos)
\end{itemize}
\end{frame}

\begin{frame}{Moving your work between the areas}
\diagram{git-workflow}

\medskip
\begin{itemize}
    \item \cmd{add}, \cmd{commit}: write your journal entry
    \item \cmd{push}: file it on GitHub, like the department archive
    \item \cmd{pull}: get what others, or you on another computer, pushed
\end{itemize}
\hint{Later, not needed today: \cmd{fetch} (download only), \cmd{switch} (change branch),
    \cmd{merge} (combine branches)}
\end{frame}

\begin{frame}[fragile]{The basic Git workflow}
Every working day: pull, edit, add, commit, push
\begin{lstlisting}
cd HPC_intro            # your clone

git pull                # get the latest changes
# ... edit your files ...
git status              # what has changed?
git add <file>          # stage changes for the commit
git commit -m "Describe what you changed"
git push                # upload your commits
\end{lstlisting}
\hint{\cmd{git push} needs your own fork and your SSH key (exercises 2.1, 2.4 and 2.7)}
\end{frame}

\begin{frame}{Your clone, the original and your fork}
\emph{A fork is like copying another lab's protocol to adapt it}

\smallskip
\diagram{git-fork}

\medskip
\begin{itemize}
    \item Fork the original on GitHub for a copy you own (exercise 2.1)
    \item Point your clone at your fork with \cmd{set-url}, then push (exercise 2.7)
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Useful Git commands for the exercises}
\small
\begin{cmdtablex}{lL}
    \textbf{git \ldots}               & \textbf{What it does}              \\
    \midrule
    \cmd{diff}                        & Show unstaged changes              \\
    \cmd{log --oneline}               & List earlier commits               \\
    \cmd{restore <file>}              & Undo unstaged changes to a file    \\
    \cmd{remote -v}                   & Show where you push and pull       \\
    \cmd{remote set-url origin <url>} & Point your clone at another remote \\
\end{cmdtablex}
\medskip
\hint{Once per computer: \cmd{git config --global user.name "<name>"}, and the same for \cmd{user.email} (exercise 2.3)}
\end{frame}
