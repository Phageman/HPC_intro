\section[Packages]{Package management}

\begin{frame}{Package managers install software without admin rights}
\begin{columns}[c]
    \column{0.7\textwidth}
    \small
    Installing tools yourself is hard:
    \begin{itemize}
        \item No admin rights on the HPC, so no \mbox{\cmd{sudo apt install}}
        \item Tools need specific versions of other software, and those versions can conflict
        \item ``It works on my computer'' is not reproducible
    \end{itemize}
    \medskip
    Conda solves this:
    \begin{itemize}
        \item Installs in your own directory
        \item Resolves dependencies for you
        \item Separate environments per project
    \end{itemize}
    \column{0.28\textwidth}
    \centering
    \includegraphics[height=1.6cm]{logo/condaforge.png}\\[0.2cm]
    \includegraphics[width=0.95\linewidth]{logo/bioconda.png}
\end{columns}
\end{frame}

\begin{frame}{Conda: packages, environments and channels}
\small
\renewcommand{\arraystretch}{1.25}
\begin{cmdtablex}{lL}
    \textbf{Term} & \textbf{What it means}                                                                                  \\
    \midrule
    Package       & A tool plus everything it needs, built for one system, e.g.\ \cmd{seqkit} for Linux                     \\
    Environment   & An isolated directory with a set of packages. Activate it to use them; other environments are not affected \\
    Channel       & An online repository of packages, e.g.\ conda-forge or Bioconda                                         \\
    Solver        & Works out which versions of all packages fit together                                                   \\
\end{cmdtablex}
\end{frame}

\begin{frame}{Install Miniforge, not Anaconda}
\small
\begin{cmdtablex}{lLll}
    \textbf{Installer}  & \textbf{Includes}      & \textbf{Channel}     & \textbf{Licence}             \\
    \midrule
    Anaconda            & Conda + extras         & defaults             & Paid if $\geq$ 200 staff$^*$ \\
    Miniconda3          & Conda                  & defaults             & Paid if $\geq$ 200 staff$^*$ \\
    \textbf{Miniforge3} & \textbf{Conda + Mamba} & \textbf{conda-forge} & \textbf{Open source, free}   \\
\end{cmdtablex}

\smallskip
\hint{$^*$Anaconda Terms of Service (2024), for its defaults channel. Conda itself is free}

\bigskip
\textbf{Mamba}: a faster Conda. Since Conda 23.10, Conda uses Mamba's solver, so both are fast now
\end{frame}

\begin{frame}[fragile]{Installing Miniforge}
On Mac, Linux and WSL, these lines download the installer for your computer and run it:
\begin{lstlisting}
m=https://github.com/conda-forge/miniforge
f=Miniforge3-$(uname)-$(uname -m).sh    # your system
curl -LO $m/releases/latest/download/$f
bash $f
\end{lstlisting}
\begin{itemize}
    \item When the installer asks to initialize Conda, type \cmd{yes}, then open a new terminal
    \item On the HPC, when it asks where to install, type \cmd{/hpc/<group>/<user>/data/miniforge3}
\end{itemize}
\hint{Copy the lines from exercise 3.1 instead of typing them}
\end{frame}

\begin{frame}[fragile]{Use conda-forge and Bioconda instead of defaults}
\small
\emph{Channels are suppliers, and defaults is the one with licence fees}
\begin{itemize}
    \item \textbf{conda-forge}: community channel with general software (Python, R, compilers, \ldots)
    \item \textbf{Bioconda}: bioinformatics tools, built on top of conda-forge.
          R packages from Bioconductor are here as \cmd{bioconductor-<name>}
    \item \textbf{defaults}: Anaconda's own channel.
          Licence restrictions, and mixing it with conda-forge can cause conflicts
\end{itemize}
Set this up once:
\begin{lstlisting}
conda config --add channels bioconda
conda config --add channels conda-forge
conda config --set channel_priority strict
\end{lstlisting}
\hint{Miniforge already uses conda-forge: the new parts are Bioconda and strict priority.
    Apple Silicon Mac: some Bioconda tools have no ARM build; use \cmd{CONDA_SUBDIR=osx-64} or the HPC}
\end{frame}

\begin{frame}[fragile]{The basic Conda workflow}
\emph{An environment is a kit for one assay, with exact lot numbers}

\smallskip
One environment per project: create it once, activate it every time
\begin{lstlisting}
conda create -n qc seqkit   # new env with seqkit
conda activate qc           # start using it
conda install fastp         # add more packages
conda list                  # what is installed here?
conda deactivate            # stop using it

conda env list              # all your environments
conda env remove -n qc      # delete an environment
# share it: the reagent list for your methods section
conda env export -n qc --from-history > environment.yml
\end{lstlisting}
\end{frame}
